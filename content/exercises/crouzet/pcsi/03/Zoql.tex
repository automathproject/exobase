\titre{Un autre produit}
\contenu{
\question{Exprimer le produit suivant à l'aide de factorielles :
\[ \prod_{k=1}^n \left(\frac12 -k\right) \]}
\reponse{On ré-écrit :
\begin{align*}
  \prod_{k=1}^n \left(\frac12 - k\right) &= \prod_{k=1}^n \frac{1-2k}{2} \\
  &=\frac{\prod\limits_{k=1}^n -(2k-1)}{\prod\limits_{k=1}^n 2}\\
  &=\frac{(-1)^n \times 1\times 3\times\ldots\times (2n-1)}{2^n}\\
  &= \frac{(-1)^n \times 1\times 2\times \ldots \times (2n-1)\times 2n}{2\times 4\times\ldots\times (2n) \times 2^n}\\
  &= \frac{(-1)^n \times (2n)!}{2^n \times 1\times 2\times \ldots\times n\times 2^n}\\
  &= \frac{(-1)^n (2n)!}{2^{2n} n!}
\end{align*}}
}
