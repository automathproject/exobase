\texte{Dans $\R^2$ euclidien, les ensembles suivants sont-ils compacts~?}
\begin{enumerate}
    \item \question{$A=\{(x,y)\in\R^2 \mid \frac12 \le \| (x,y)\| \le 2\ \text{et}
\ xy=1\}$.}
    \item \question{$B=\{(x,y)\in\R^2 \mid \frac12 < \| (x,y)\| \le 2\ \text{et}
\ xy=1\}$.}
    \item \question{$C= \{(x,\cos n)\in\R^2 \mid 0\le x \le 18 \ \text{et}\ n\in\N\}$.}
\end{enumerate}
