\contenu{
\texte{Soient $A$ et $B$ deux parties non vides et majorées de $\R$. Montrer
les implications suivants :}
\begin{enumerate}
    \item \question{$\exists M \in\R\;\forall x\in A,\; x<M\Rightarrow\sup A\leq M$}
    \item \question{$A\subset B\Rightarrow\sup A\leq\sup B$.}
\end{enumerate}
}
