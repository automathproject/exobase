\begin{enumerate}
    \item \question{On considère l'application $\xi_s(q)=-sqs^{-1}\ (s\in(\R^3)^*,\
  q\in\R^3).$ Montrer que  $\xi_s$ est la reflection dans
  $\R^3$  par rapport au plan $s^\bot.$}
    \item \question{Démontrer la surjectivité de l'application $\varphi :  S^3\to SO(3)$ où
$\varphi(s)=\rho_s,$ et $\rho_s(q)=sqs^{-1}$ ($s\in S^3,\ q\in\R^3$).}
\end{enumerate}
