\contenu{
\question{
Trouver le commutant de $\begin{pmatrix}2&-2&1\\ 2&-3&2\\ 1&2&0\end{pmatrix}$.
}
\reponse{
3 valeurs propres distinctes, $M$ est diagonalisable et son
	     commutant est l'ensemble des polynômes en $M$ :
	     $aI + bM + cM^2$, $a,b,c\in K$.
}
}
