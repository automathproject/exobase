\contenu{
\question{
Soit $(S) \iff AX = B$ un système linéaire de $n$ équations à $n$ inconnues
de Cramer.
Montrer que pour tous scalaires $c_1,\dots,c_n$, on a :
$$c_1x_1 + \dots + c_nx_n =
  -\frac1{\det A}\left|\,
                  \begin{array}{ccc|c}
                                &      &                         &b_1     \\
                                &A     &                         &\vdots  \\
                                &      &                         &b_n     \\ \hline
                            c_1 &\dots &c_n                      &0       \end{array}
                  \,\right|.$$
}
}
