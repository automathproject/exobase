\contenu{
\texte{Soit $G$ le groupe multiplicatif des matrices complexes
d'ordre $n$. Parmi les sous-ensembles suivants de $G$, lesquels
sont des sous-groupes ?}
\begin{enumerate}
    \item \question{[---] les matrices \`a coefficients r\'eels\,;}
    \item \question{[---] les matrices inversibles\,;}
    \item \question{[---] les matrices r\'eelles inversibles \`a coefficients
positifs\,;}
    \item \question{[---] les matrices diagonales inversibles\,;}
    \item \question{[---] les matrices v\'erifiant $a_{i,i} \neq 0, \forall i$,  et
triangulaires sup\'erieures ($a_{i,j} = 0$ si $i > j$).}
    \item \question{[---] les matrices v\'erifiant $a_{i,j}= \overline{a_{j,i}},
\forall i,j$.}
\end{enumerate}
}
