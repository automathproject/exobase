\texte{
Soient $A=\left(\begin{smallmatrix}
1&1\\0&1\end{smallmatrix}\right)$ et
$C=\left\{M\in\mathcal{M}_2(\Rr): MA=AM\right\}.$
}
\begin{enumerate}
    \item \question{Montrer que $C$ est un sous-espace vectoriel de $\mathcal{M}_2(\Rr)$
 et en d\'eterminer une base.}
    \item \question{Montrer que, pour les lois usuelles, $C$ est une $\Rr$-alg\`ebre.}
\end{enumerate}
