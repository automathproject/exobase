\module{Géométrie}

\difficulte{}

\contenu{
\texte{
Soit 
$P$ un plan muni d'un rep\`ere $(O,{\overset{\rightarrow}{i}},
{\overset{\rightarrow}{j}})$ quelconque.
}
\texte{On consid\`ere $D$ une droite d'\'equation cart\'esienne 
$2x-y+3=0$ et ${\overset{\rightarrow}{u}}(3,-2).$}
\begin{enumerate}
    \item \question{Soit $A(4,2).$ Donner une \'equation param\'etrique de 
$D_A$ droite passant par $A$ de direction ${\overset{\rightarrow}{u}}.$ 
En d\'eduire les coordonn\'ees de $A'=D_A\cap D$ projet\'e de 
$A$ sur $D$ selon ${\overset{\rightarrow}{u}}.$}
    \item \question{D\'efinir plus g\'en\'eralement analytiquement la projection sur 
$D$ selon ${\overset{\rightarrow}{u}}$ en exprimant 
les coordonn\'ees $x',$ $y'$ de $M'$ projet\'e de $M(x,y)$ 
en fonction de $x$ et $y.$\\}
\end{enumerate}
}
