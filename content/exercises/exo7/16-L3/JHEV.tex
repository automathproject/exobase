\texte{
On rappelle que pour tout entier naturel $n$, et tout nombre complexe $z$, $n^z$ désigne $p_z(n)=\exp(z\log n)$.
}
\begin{enumerate}
    \item \question{Déterminer le module de $n^z$.}
    \item \question{Montrer que la série $\sum_{n\geq 1} \frac{1}{n^z}$ converge normalement sur l'ouvert $\{z\in\Cc / re(z)>1\}$.}
\end{enumerate}
