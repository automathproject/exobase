\contenu{
\texte{
Lesquels des ensembles de nombres suivants sont des groupes ?
}
\begin{enumerate}
    \item \question{Les nombres irrationnels munis de l'addition ; de la multiplication ;}
    \item \question{Les nombres complexes de valeur absolue $1$ munis de l'addition ; de la multiplication ;}
    \item \question{Les nombres complexes munis de l'opération binaire $z*z'=|z|\cdot z'$.}
\end{enumerate}
