\question{
Convergence et calcul de :
\[
\int_{0}^{1}\frac{\ln (1+t^{2})dt}{t^{2}}, \qquad
\int_{0}^{\infty }\ln \left( 1+\frac{1}{t^{2}}\right) dt, \qquad
\int_{1}^{\infty }\frac{\ln t}{t^{n}}dt.
\]
}
\reponse{
R\'{e}ponses : $\frac{\pi }{2}-\ln 2,\pi ,\frac{1}{\left( n-1\right) ^{2}}$.
}
