\contenu{
\question{
Calculer $\mathrm{cotan} x - 2\mathrm{cotan}2x$.
Simplifier $\sum_{k=0}^n \frac 1{2^k}\tan\frac\alpha{2^k}$.
}
\reponse{
$\mathrm{cotan} x - 2\mathrm{cotan}2x = \tan x$,
         $\sum = \frac 1{2^n}\mathrm{cotan}\frac\alpha{2^n} - 2\mathrm{cotan}2\alpha$.
}
}
