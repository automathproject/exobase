\begin{enumerate}
    \item \question{Soit $(u_n)$ une suite r{\'e}elle telle que $\frac {u_n}{1+u_n} \xrightarrow[n\to\infty]{} 0$.
    Montrer que $u_n \xrightarrow[n\to\infty]{} 0$.}
    \item \question{M{\^e}me question avec
    $\begin{cases} \frac {u_n}{1+u_n^2} \xrightarrow[n\to\infty]{} 0\cr
             (u_n) \text{ est born{\'e}e.}\cr\end{cases}$}
\end{enumerate}
