\texte{
Effectuer les calculs suivants :
}
\begin{enumerate}
    \item \question{$(3+2i)(1-3i)$.}
    \item \question{Produit du nombre complexe de module $2$ et d'argument $\pi /3$ par le
nombre complexe de module $3$ et d'argument $-5\pi /6$.}
    \item \question{$\frac{3+2i}{1-3i}$.}
    \item \question{Quotient du nombre complexe de module $2$ et d'argument $\pi /3$ par le
nombre complexe de module $3$ et d'argument $-5\pi /6$.}
\reponse{
$9-7i$; \qquad
 $-6i$; \qquad
 $-0{,}3+1{,}1i$; \qquad
 $-\frac{\sqrt3}{3}-\frac{i}{3}$.
}
\end{enumerate}
