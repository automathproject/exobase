\contenu{
\question{
Pour quels $n\in \N$ le polyn{\^o}me $(1+X^4)^n - X^n$ est-il divisible par
    $1+X+X^2$ dans $\R[X]$ ?
}
\reponse{
$n\equiv 0(\mathrm{mod}\, 6)$.
}
