\contenu{
\question{
Soit $P = X^5 + aX^2 + 15X - 6i$. Trouver $a \in \C$ tel que
$P$ a une racine triple dans $\C$. Factoriser alors $P$.
}
\reponse{
$a = 10i$.
Racines : $i,i,i,\frac {-3i+\sqrt{15}}2, \frac {-3i-\sqrt{15}}2$.
}
