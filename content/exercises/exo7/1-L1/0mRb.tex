\contenu{
\question{
Existe-t-il deux matrices carrées $A$ et $B$ telles que $AB-BA=I_n$.
}
\reponse{
Non, car $\text{Tr}(AB-BA)=\text{Tr}(AB) -\text{Tr}(BA) = 0\neq n =\text{Tr}(I_n)$.
}
}
