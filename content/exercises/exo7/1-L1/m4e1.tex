\contenu{
\question{
Soit $f$ une application lin\'eaire de $\Rr^n$ dans $\Rr^n$. 
Montrer que les 
propri\'et\'es $(1)$ \`a $(3)$ sont \'equivalentes.
\begin{equation*}\Rr^n=Im (f) \bigoplus Ker (f)\tag*{$(1)$}\end{equation*}
\begin{equation*}Im (f)= Im (f^2)\tag*{$(2)$}\end{equation*}
\begin{equation*}Ker (f)= Ker (f^2)\tag*{$(3)$}\end{equation*}
}
}
