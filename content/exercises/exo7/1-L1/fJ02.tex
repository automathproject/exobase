\contenu{
\question{
Soit $a\in K$ et $ \Phi : { K_n[X]} \to { K_n[X]}, P \mapsto {(X-a)(P'(X)+P'(a)) + P(X)-P(a).}$

Chercher $\mathrm{Ker} \Phi$ et $\Im \Phi$.
}
\reponse{
$\mathrm{Ker}\Phi =  K_0[X]$, $\Im\Phi = (X-a) K_{n-1}[X]$.
}
