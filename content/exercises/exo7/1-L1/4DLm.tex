\contenu{
\question{
Soit $z \in \mathbb{U}$. Peut-on trouver $a \in \R$ tel que $z = \frac {1+ia}{1-ia}$ ?
}
\reponse{
$z = e^{i\theta}  \Rightarrow  a = \tan\frac\theta2$
         pour $\theta \not \equiv \pi (\mathrm{mod}\,{2\pi})$.
}
}
