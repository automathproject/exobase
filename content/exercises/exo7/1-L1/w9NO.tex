\contenu{
\texte{
Soit $E = H \oplus K$ et $(\vec e_1,\dots,\vec e_k)$ une base de $K$.
}
\begin{enumerate}
    \item \question{Montrer que pour tout $\vec a \in H,\
    K_{\vec a} = \text{vect}(\vec e_1 + \vec a, \dots, \vec e_k + \vec a)$
    est un supplémentaire de $H$.}
    \item \question{Montrer que si $\vec a \ne \vec b$, alors $K_{\vec a} \ne K_{\vec b}$.}
\end{enumerate}
