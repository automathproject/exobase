\contenu{
\question{
Prouver que $B$ divise $A$, où : \\
$A=X^{3n+2}+X^{3m+1}+X^{3p} \text{ et } B=X^2+X+1$,\\
$A=(X+1)^{2n}-X^{2n}-2X-1 \text{ et } B = X(X+1)(2X+1)$,\\
$A=nX^{n+1}-(n+1)X^n+1 \text{ et } B=(X-1)^2$.
}
