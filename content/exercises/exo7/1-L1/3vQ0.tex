\texte{
Donner la négation des phrases suivantes
}
\begin{enumerate}
    \item \question{$x\geq3$}
    \item \question{$0<x\leq2$.}
\reponse{
Le contraire de $x\geq3$ est $x<3$. Le contraire de $0<x\leq 2$ est $((x\leq0)\;\mbox{ou}\;x>2)$.
}
\end{enumerate}
