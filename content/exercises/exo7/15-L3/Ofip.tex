\texte{
Soit $E$ un espace vectoriel normé, et $(a,a')\in E^2$,
$(r,r')\in(\R_+^*)^2$. Montrer :
}
\begin{enumerate}
    \item \question{$B(a,r)=\{a\}+B(0,r)$}
    \item \question{$B(a,r)=B(a',r') \Leftrightarrow a=a' \text{ et } r=r'$}
    \item \question{$B(a+a',r+r')=B(a,r)+B(a',r')$}
    \item \question{$B(a,r)\cap B(a',r')\neq\emptyset\Leftrightarrow\|{a'-a}\|{}<r+r'$.}
\end{enumerate}
}
