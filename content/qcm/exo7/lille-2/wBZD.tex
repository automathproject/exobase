\begin{question}
\qclassification{127.11}
\qtags{motcle=changement de variables}
Le changement de variable $u=\mathrm{e}^x$ donne :
\begin{answers}
\good{$\displaystyle \int \frac{\mathrm{e}^x\mathrm{d}x}{1+\mathrm{e}^{2x}}=\int \frac{\mathrm{d}u}{1+u^2}$.}
\good{$\displaystyle \int \frac{\mathrm{e}^x\mathrm{d}x}{1+\mathrm{e}^{2x}}=\arctan(u)+k$, $k\in \Rr$.}
\bad{$\displaystyle \int \frac{\mathrm{d}x}{1+\mathrm{e}^{2x}}=\int \frac{\mathrm{d}u}{1+u^2}$.}
\bad{$\displaystyle \int \frac{\mathrm{d}x}{1+\mathrm{e}^{2x}}=\arctan(u)+k$, $k\in \Rr$.}
\end{answers}
\vskip2mm
\begin{explanations}
Avec $u=\mathrm{e}^x$, on aura : $\mathrm{d}u=\mathrm{e}^x\mathrm{d}x$. D'où
$$\displaystyle \int \frac{\mathrm{e}^x\mathrm{d}x}{1+\mathrm{e}^{2x}}=\int \frac{\mathrm{d}u}{1+u^2}=\arctan(u)+k,\; k\in \Rr.$$
\end{explanations}
\end{question}
