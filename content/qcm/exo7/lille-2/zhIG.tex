\begin{question}
\qclassification{127.11}
\qtags{motcle=primitives usuelles}
Parmi les affirmations suivantes, cocher celles qui sont vraies :
\begin{answers}
\good{$\displaystyle \int \cos x\sin x\mathrm{d}x=\frac{1}{2}\sin ^2x+k$, où $k\in \Rr$.}
\bad{$\displaystyle \int \cos x\sin x\mathrm{d}x=\frac{1}{2}\cos ^2x+k$, où $k\in \Rr$.}
\good{$\displaystyle \int \cos x\sin ^2x\mathrm{d}x=\frac{1}{3}\sin ^3x+k$, où $k\in \Rr$.}
\bad{$\displaystyle \int \cos ^2x\sin ^2x\mathrm{d}x=\frac{1}{4}\sin ^4x+k$, où $k\in \Rr$.}
\end{answers}
\begin{explanations}
Avec $u=\sin x$, on a : $\mathrm{d}u=\cos x\mathrm{d}x$,
$$\int \cos x\sin x\mathrm{d}x=\int u\mathrm{d}u=\frac{1}{2}u^2+k=\frac{1}{2}\sin ^2x+k,\; k\in \Rr,$$
et
$$\int \cos x\sin ^2x\mathrm{d}x=\int u^2\mathrm{d}u=\frac{1}{3}u^3+k=\frac{1}{3}\sin ^3x+k,\; k\in \Rr.$$
\end{explanations}
\end{question}
