\begin{question}
\qclassification{225.01}
Parmi les fonctions suivantes, quelles sont celles solutions de l'équation différentielle $y' = 3y$ ?
\begin{answers}
  \bad{$f(x) = 3e^{2x}$}
  \good{$f(x) = 2e^{3x}$}
  \bad{$f(x) = e^{-3x}$}
  \bad{$f(x) = e^{-2x}$}
\end{answers}
\begin{explanations}
La forme générale des solutions est $y(x) = Ce^{3x}$ où $C$ est une constante réelle.
\end{explanations}
\end{question}
