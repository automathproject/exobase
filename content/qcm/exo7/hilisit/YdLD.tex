\begin{question}
\qclassification{103.99}
 Quel est le chiffre des unités de $7^{100}$ ?
    \begin{answers}
        \good{$1$}
        \bad{$3$}
        \bad{$5$}
        \bad{$9$}
    \end{answers}
    \begin{explanations}
    Le chiffre des unités est donné par la congruence modulo $10$.\\
    Puisque $7^2 = 49 \equiv (-1) \;[10]$, on a :
    $$ 7^{100} = (7^2)^{50} \equiv (-1)^{50} \equiv 1 \;[10]$$
    \end{explanations}
\end{question}
