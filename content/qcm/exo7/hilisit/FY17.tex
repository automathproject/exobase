\begin{question}
\qclassification{103.99}
     Soient $a\equiv 2 \;[13]$ et $b \equiv 7 \; [13]$.
    Quelles sont les affirmations vraies ?
    \begin{answers}
        \good{$a+b \equiv 9 \;[13]$}
        \good{$ab \equiv 1 \; [13]$}
        \good{$a^2 \equiv -9 \;[13]$}
        \good{$b^3 \equiv 5 \;[13]$}
    \end{answers}
    \begin{explanations}
        Tout est vrai ! Modulo $13$, on a bien :
        $$a+b = 2+7 = 9 \equiv 9,$$
        $$ab = 2 \times 7 = 14 \equiv 1,$$
        $$a^2 = 2^2 = 4 \equiv -9,$$
        $$b^3 = 7^3 = 343 \equiv 5.$$
    \end{explanations}
\end{question}
