\contenu{
\begin{question}
\qclassification{103.04}
    Les entiers suivants sont-ils des nombres premiers ?
    \begin{answers}
        \good{$107$}
        \good{$113$}
        \bad{$145$}
        \bad{$153$}
    \end{answers}
    \begin{explanations}
        $107$ et $113$ sont des nombres premiers ;
        $145 = 5 \times 29$ ;  $153 = 3^2 \times 17$.
    \end{explanations}
\end{question}
