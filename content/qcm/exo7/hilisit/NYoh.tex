\begin{question}
\qclassification{100.02}
Parmi ces ensembles, quels sont ceux qui sont inclus dans $\{ 0 , 1 , 2 \}$ ?
\begin{answers}
	\good{$\{ x \in \Rr \; | \; x(x-2)=0 \}$}
	\good{$\{ x \in \Rr \; | \; e^x = 1 \}$}
	\bad{$\{ x > 0 \; | \; \ln(x) = 1 \}$}
	\good{$[0,3] \cap \{ x \in \Rr \; | \; (x+2)^2 = 4  \}$}
\end{answers}
\begin{explanations}
$\{ x \in \Rr \; | \; x(x-2)=0 \} = \{0,2\}$.\\
$\{ x \in \Rr \; | \; e^x = 1 \} = \{0\}$.\\
$\{ x > 0 \; | \; \ln(x) = 1 \} = \{ e \}$.\\
$[0,3] \cap \{ x \in \Rr \; | \; (x+2)^2 = 4  \}$ = $[0,3] \cap \{ -4 , 0\} = \{0\}$.
\end{explanations}
\end{question}
