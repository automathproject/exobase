\begin{question}
\qclassification{103.03}
    On considère $a,b$ et $d$ des entiers tels que $d | a$ et $d | b$.
    Quelles sont les affirmations vraies ?
    \begin{answers}
        \good{$d | a+b$}
        \good{$d | a-b$}
        \good{$d | a \times b$}
        \bad{$d | \frac{a}{b}$}
    \end{answers}
    \begin{explanations}
        L'affirmation $d | \frac{a}{b}$ est fausse et n'a même pas toujours de sens. Le reste est vrai.
    \end{explanations}
\end{question}
