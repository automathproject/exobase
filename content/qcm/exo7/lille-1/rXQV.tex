\begin{question}
\qclassification{120.04}
\qtags{motcle=intervalle}
Quelles sont les assertions vraies ?
\begin{answers}
    \bad{$[3,7] \cup [8,10] = [3,10]$}

    \bad{$[-3,5] \cap [2,7] = [-3,7]$}

    \bad{$[a,b[\cup ]a,b] = ]a,b[$}

    \good{$[a,b[\cap ]a,b] = ]a,b[$}
\end{answers}
\begin{explanations}
Tracer les intervalles sur la droite réelle pour mieux comprendre.
\end{explanations}
\end{question}
