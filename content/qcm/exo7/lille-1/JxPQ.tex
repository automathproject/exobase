\begin{question}
\qclassification{100.01}
\qtags{motcle=vrai/faux}
Soit $P$ une assertion vraie et $Q$ une assertion fausse. Quelles sont les assertions vraies ?
\begin{answers}
    \good{$P$ ou $Q$}

    \bad{$P$ et $Q$}

    \bad{non($P$) ou $Q$}

    \good{non($P$ et $Q$)}
\end{answers}
\begin{explanations}
$P$ ou $Q$ est vraie. Comme $P$ et $Q$ est fausse alors non($P$ et $Q$) est vraie.
\end{explanations}
\end{question}
