\begin{question}
\qclassification{100.01}
\qtags{motcle=implication/équivalence}
À quoi est équivalent $P \implies Q$ ?
\begin{answers}
    \bad{non($P$) ou non($Q$)}

    \bad{non($P$) et non($Q$)}

    \good{non($P$) ou $Q$}

    \bad{$P$ et non($Q$)}
\end{answers}
\begin{explanations}
La définition (à connaître) de "$P \implies Q$" est "non($P$) ou $Q$".
\end{explanations}
\end{question}
