\begin{question}
\qclassification{100.02,101.01,102.01,102.02}
\qtags{motcle=union}
Soit $A=[-1,3]$ et $B=[0,4]$. Cochez la bonne réponse :
\begin{answers}
    \bad{$A\cup B=\varnothing$}
    \bad{$A\cup B=[0,3]$}
    \bad{$A\cup B=[-1,0]$}
    \good{$A\cup B=[-1,4]$}
\end{answers}
\begin{explanations}
L'ensemble $A\cup B$ est formé des éléments qui sont dans $A$ ou dans $B$.
\end{explanations}
\end{question}
