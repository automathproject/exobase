\begin{question}
\qclassification{100.99}
Soit $f:\Rr \to \Rr$, $x \mapsto x^2+1$.
\begin{answers}
    \bad{$f$ est injective.}

    \good{$f$ n'est pas injective.}

    \bad{$f$ est surjective.}

    \good{$f$ n'est pas surjective.}

    \good{La restriction de $f$, $f_| : [1,2] \to [2,5]$ est bijective.}
\end{answers}
\end{question}
