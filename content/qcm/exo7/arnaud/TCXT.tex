\begin{question}
\qclassification{100.99}
Pour $x,y \in \Rr$ et $z=x+iy$, on pose $e^z=e^x \times
e^{iy}=e^{x+iy}$.
\begin{answers}
    \good{$|e^z|=e^x$.}

    \bad{$|e^z|= \sqrt{x^2+y^2}$.}

    \good{Arg $e^z = y$.}

    \bad{Arg $e^z=x+y$.}

    \bad{La fonction $f:\Cc \to \Cc$, $z \mapsto e^z$ est injective.}
\end{answers}
\end{question}
